106 ", max memory to use = ", _max_memory_to_use));

108 // first permutation
109 dat::permutation _permutation;
110 for (dat::entries::const_entry_ite _entry = _entries->begin();
111 _entry != _entries->end(); ++_entry) {
112 _permutation.push_back(_entry);
113 }

116
117 const auto _initial = std::chrono::high_resolution_clock::now();

121 format::bus::fmt("initial = ", _initial.time_since_epoch().count()));

123 std::size_t _slept{0};
124 const auto _amount_to_sleep{100ms};

128 break;
129 }

147 _result != async::dat::result::OK) {

150 format::bus::fmt(
151 "Error publishing 'internal::grid_permutations_tried'",
152 _result));
153 }
154 }
155
156 if (should_break(_initial, _permutation_counter, p_max_permutations)) {
157 break;
158 }
159
160 if (auto _result{
161 m_internal_dispatcher
162 .template publish<evt::internal::grid_to_organize>(_grid)};
163 _result != async::dat::result::OK) {
164 TNCT_LOG_ERR(
165 m_logger,
166 format::bus::fmt("Error publishing 'grid_to_organize'", _result));
167 }
168
169 if (should_break(_initial, _permutation_counter, p_max_permutations)) {
170 break;
171 }
172
173 std::unique_lock<std::mutex> _lock_wait(m_mutex_wait);
174 m_cond_wait.wait_for(_lock_wait, _amount_to_sleep, [&]() {
175 if (!should_break(_initial, _permutation_counter, p_max_permutations) &&
176 should_wait(_max_memory_to_use)) {
177 ++_slept;
178 return false;
179 }
180 return true;
181 });
182
183 if (should_break(_initial, _permutation_counter, p_max_permutations)) {
184 break;
185 }
186
187 std::next_permutation(_permutation.begin(), _permutation.end(),
188 [this](dat::entries::const_entry_ite p_e1,
189 dat::entries::const_entry_ite p_e2) -> bool {

191 });

195 auto _elapsed =
196 std::chrono::duration_cast<std::chrono::seconds>(_current - _initial);
197
198 TNCT_LOG_DEB(m_logger,
199 format::bus::fmt(
200 "left the loop, solved = ", (m_solved ? "Y" : "N"),
201 ", timeout = ", (m_timeout ? "Y" : "N"),
202 ", stop = ", (m_stop ? "Y" : "N"), ", waiting more ",
203 (m_wait_for +
204 std::chrono::seconds(_slept * _amount_to_sleep.count())) -
205 _elapsed,
206 " secs"));
207
208 std::unique_lock<std::mutex> _lock_finish(m_mutex_finish);
209 m_cond_finish.wait_for(
210 _lock_finish,
211 (m_wait_for + std::chrono::seconds(_slept * _amount_to_sleep.count())) -
212 _elapsed,
213 [&]() { return (m_solved || m_timeout || m_stop); });
214
215 TNCT_LOG_DEB(m_logger, "left the condition");
216
217 if (m_solved) {
218 TNCT_LOG_DEB(m_logger, "grid solved");
219
220 m_internal_dispatcher.template clear<evt::internal::grid_to_organize>();
221 return m_solved;
222 }
223
224 if (m_timeout) {
225 TNCT_LOG_DEB(m_logger, "timeout");
226
227 m_internal_dispatcher.template clear<evt::internal::grid_to_organize>();
228 }
229
230 return {};

413 void on_new_grid_to_organize(evt::internal::grid_to_organize &&p_event) {
414 if (p_event.grid == nullptr) {
415 TNCT_LOG_ERR(m_logger,
416 "could not call organizer because p_event.grid is null");
417 return;
418 }
419 if (m_solved) {
420 m_cond_wait.notify_all();

423 // internal::organizer _organizer{m_dispatcher};
424
425 TNCT_LOG_TRA(m_logger,
426 format::bus::fmt("calling organizer for permutation ",
427 p_event.grid->get_permutation_number()));
428
429 if (m_organizer(p_event.grid)) {
430
431 TNCT_LOG_DEB(m_logger,
432 format::bus::fmt("grid was organized at permutation ",
433 p_event.grid->get_permutation_number()));
434
435 // m_dispatcher. template clear<evt::internal::grid_to_organize>();
436 m_solved = p_event.grid;
437 m_cond_wait.notify_all();
438 m_cond_finish.notify_one();
439 return;
440 }
441
442 // if assembler is waiting for an available slot, in case memory is full, we
443 // notify that at one permutation was removed from the queues, but only if
444 // the grid was not solved by another organizer
445 if (!m_solved) {
446 m_cond_wait.notify_all();
