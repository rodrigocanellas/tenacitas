352 std::lock_guard<std::mutex> _lock{m_mutex};

354 m_table.push_back({std::move(p_object)});
355
356 record_ref _record_ref{*std::prev(m_table.end())};
357
358 bool _error{false};
359 auto _visitor{[&]<tuple::cpt::is_tuple t_tuple, std::size_t t_field_pos>() {
360 if constexpr (is_index<t_field_pos>()) {
