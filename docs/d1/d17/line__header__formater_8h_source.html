27static inline std::stringstream &
28default_line_header_formater(std::stringstream &p_stream, dat::level p_level,
29 const std::source_location &p_source_location)
30{
31 static constexpr char _separator = '|';
32 static constexpr uint8_t _max_file_name_size = 35;
33
34 const double _now_microsecs =
35 std::chrono::duration_cast<std::chrono::microseconds>(
36 std::chrono::system_clock::now().time_since_epoch())
37 .count();
38 const time_t _now_seconds{static_cast<time_t>(_now_microsecs / 1000000)};
39
40 // _time_str must have space enough for YYYY-MM-DD HH:MM:SS
41 char _time_str[4 + 1 + 2 + 1 + 2 + 1 + 2 + 1 + 2 + 1 + 2 + 1 + 1];
42 strftime(_time_str, sizeof _time_str, "%Y-%m-%d %H:%M:%S",
43 localtime(&_now_seconds));
44
45 auto _aux(
46 std::filesystem::path(p_source_location.file_name()).filename().string());
47
48 decltype(_aux) _file_name;
49
50 if (_aux.size() > _max_file_name_size)
51 {
52 _file_name.append(&_aux[0], &_aux[_max_file_name_size - 3]);
53 _file_name.append("...");
54 }
55 else
56 {
57 _file_name = _aux;
58 }
59
60 p_stream
61 << p_level << _separator << _time_str << ','
62 << format::bus::format_fix_number(
63 static_cast<uint32_t>(_now_microsecs - (_now_seconds * 1000000)),
64 6, '0', tnct::format::dat::align::right)
65 << _separator << std::this_thread::get_id() << _separator
66 << std::setfill(' ') << std::left << std::setw(_max_file_name_size)
67 << _file_name << _separator
68 << format::bus::format_fix_number(p_source_location.line(), uint8_t{5})
69 // << _separator << p_source_location.function_name()
70
71 << _separator;
72 return p_stream;
73}
