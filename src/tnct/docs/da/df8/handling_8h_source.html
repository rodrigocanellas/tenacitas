302 // Waits for an event to be available in the event queue, then remove it and
303 // call the handler in \p p_handling_handler_pos in \p m_handling_handlers. It
304 // exits when \p m_stop is set.
305 void handler_loop(handling_handler_pos p_handling_handler_pos)
306 {
307
308 // auto _queue_addr{&m_queue};
309 auto _loop_id{std::this_thread::get_id()};
310
311 TNCT_LOG_TRA(
312 m_logger,
313 format::bus::fmt("p_handling_handler_pos = ", p_handling_handler_pos,
314 " ", trace("starting subscriber's loop", _loop_id)));
315
316 // auto _subscriber_id{&m_handling_handlers[p_handling_handler_pos]};
317
318 if (m_stopped)
319 {
320 TNCT_LOG_TRA(m_logger, trace("already stopped", _loop_id));
321 return;
322 }
323
324 TNCT_LOG_TRA(m_logger, trace("entering loop"));
325
326 while (true)
327 {
328 {
329 TNCT_LOG_TRA(m_logger, trace("locking", _loop_id));
330
331 std::unique_lock<std::mutex> _lock(m_data_mutex);
332 m_data_cond.wait(
333 _lock,
334 [this, /*_subscriber_id,*/ _loop_id /*,_queue_addr*/]() -> bool
335 {
336 TNCT_LOG_TRA(m_logger, trace("entering condition", _loop_id));
337
338 if (m_stopped)
339 {
340 TNCT_LOG_TRA(m_logger, trace("stopped", _loop_id));
341 return true;
342 }
343 if (!m_queue.empty())
344 {
345 TNCT_LOG_TRA(m_logger, trace("there is data", _loop_id));
346 return true;
347 }
348 TNCT_LOG_TRA(m_logger, trace("waiting", _loop_id));
349 return false;
350 });
351 }
352
353 TNCT_LOG_TRA(m_logger, trace("lock released", _loop_id));
354
355 if (m_stopped)
356 {
357 TNCT_LOG_TRA(m_logger, trace("stopped due to explicit stop", _loop_id));
358 break;
359 }
360
361 TNCT_LOG_TRA(m_logger, trace("getting event from the queue", _loop_id));
362
363 std::optional<event> _maybe{m_queue.pop()};
364 if (!_maybe.has_value())
365 {
366 TNCT_LOG_TRA(m_logger, trace("no event in queue", _loop_id));
367 continue;
368 }
369 event _event{std::move(*_maybe)};
370
371 TNCT_LOG_TRA(
372 m_logger,
373 format::bus::fmt("p_handling_handler_pos = ", p_handling_handler_pos,
374 ", handler address = ",
375 &m_handling_handlers[p_handling_handler_pos], " ",
376 trace("about to call subscriber", _loop_id)));
377
378 m_handling_handlers[p_handling_handler_pos](std::move(_event));
379
380 TNCT_LOG_TRA(m_logger, trace("event handled", _loop_id));
381 }
382
383 TNCT_LOG_TRA(m_logger, trace("leaving subscriber's loop", _loop_id));
384 }
