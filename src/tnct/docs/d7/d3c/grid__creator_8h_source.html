299 void operator()(const dat::entries &p_entries, dat::index p_num_rows,
300 dat::index p_num_cols, std::chrono::seconds p_wait_for,
301 dat::index p_max_num_rows)
302 {
303
304 try
305 {
306
307 if (p_wait_for <= std::chrono::seconds::zero())
308 {
309 if (const auto _result =
310 m_dispatcher.template publish<evt::internal::grid_create_error>(
311 dat::error::INVALID_INTERVAL);
312 _result != async::dat::result::OK)
313 {
314 TNCT_LOG_ERR(m_logger,
315 format::bus::fmt("error publishing error ",
316 dat::error::INVALID_INTERVAL));
317 }
318 return;
319 }
320
321 if (p_max_num_rows <= p_num_rows)
322 {
323 if (const auto _result =
324 m_dispatcher.template publish<evt::internal::grid_create_error>(
325 dat::error::MAX_ROWS_SMALLER_THAN_ROWS);
326 _result != async::dat::result::OK)
327 {
328 TNCT_LOG_ERR(m_logger, format::bus::fmt(
329 "error publishing error ",
330 dat::error::MAX_ROWS_SMALLER_THAN_ROWS));
331 }
332 return;
333 }
334
335 if (std::optional<dat::entries::const_entry_ite> _entry{
336 is_any_word_too_long(p_entries, p_num_rows, p_num_cols)};
337 _entry.has_value())
338 {
339 if (const auto _result =
340 m_dispatcher.template publish<evt::internal::grid_create_error>(
341 dat::error::WORD_TOO_LONG, _entry.value()->get_word());
342 _result != async::dat::result::OK)
343 {
344 TNCT_LOG_ERR(m_logger, format::bus::fmt("error publishing error ",
345 dat::error::WORD_TOO_LONG));
346 }
347 return;
348 }
349
350 m_stop = false;
351
352 const auto _max_permutations = number_of_permutations(p_entries);
353
354 crosswords::dat::index _current_num_rows{p_num_rows};
355 crosswords::dat::index _current_num_cols{p_num_cols};
356 m_dispatcher.clear();
357
358 while ((_current_num_rows <= p_max_num_rows) && (!m_stop))
359 {
360
361 TNCT_LOG_DEB(m_logger,
362 format::bus::fmt("Trying grid ", _current_num_rows, 'x',
363 _current_num_cols));
364
365 auto _start = std::chrono::high_resolution_clock::now();
366
367 std::shared_ptr<crosswords::dat::grid> _grid{
368 m_assembler.start(p_entries, _current_num_rows, _current_num_cols,
369 p_wait_for, _max_permutations)};
370
371 if (_grid)
372 {
373
374 auto _end = std::chrono::high_resolution_clock::now();
375
376 std::chrono::duration<double> _diff = _end - _start;
377
378 TNCT_LOG_DEB(m_logger,
379 format::bus::fmt("grid ", _current_num_rows, 'x',
380 _current_num_cols, " solved in ",
381 _diff.count(), " seconds"));
382
383 if (const auto _result{
384 m_dispatcher
385 .template publish<evt::internal::grid_create_solved>(
386 _grid,
387 std::chrono::duration_cast<std::chrono::seconds>(
388 std::chrono::duration<double>(_diff)),
389 _max_permutations)};
390 _result != async::dat::result::OK)
391 {
392 TNCT_LOG_ERR(m_logger,
393 format::bus::fmt(
394 "Error publishing internal::grid_create_solved: ",
395 _result));
396 }
397
398 TNCT_LOG_DEB(m_logger, "evt::internal::grid_create_solved published");
399
400 return;
401 }
402
403 if (m_stop)
404 {
405 break;
406 }
407
408 TNCT_LOG_DEB(m_logger,
409 format::bus::fmt("Not solved for ", _current_num_rows, 'x',
410 _current_num_cols));
411
412 ++_current_num_rows;
413 ++_current_num_cols;
414
415 m_dispatcher.clear();
416 }
417
418 TNCT_LOG_DEB(m_logger, "Not solved for all the grid sizes");
419
420 if (const auto _result{
421 m_dispatcher
422 .template publish<evt::internal::grid_create_unsolved>(
423 _current_num_rows, _current_num_cols)};
424 _result != async::dat::result::OK)
425 {
426 TNCT_LOG_ERR(
427 m_logger,
428 format::bus::fmt(
429 "Error publishing internal::grid_create_unsolved: ", _result));
430 }
431 m_dispatcher.clear();
432 }
433 catch (dat::error &_error)
434 {
435 if (const auto _result{
436 m_dispatcher.template publish<evt::internal::grid_create_error>(
437 _error)};
438 _result != async::dat::result::OK)
439 {
440 TNCT_LOG_ERR(m_logger,
441 format::bus::fmt("could not publish the error ", _error))
442 }
443 }
444 catch (std::exception &_ex)
445 {
446 TNCT_LOG_ERR(m_logger, format::bus::fmt("error '", _ex.what(), '\''));
447 }
448 catch (...)
449 {
450 TNCT_LOG_ERR(m_logger, "unknown error");
451 }
